### 証拠品の「！」判定

```js
function evidenceHasNewInfo(item) {
  if (!state.visitedEvidence.includes(item.id) ||
      hasNewUnlock(`evidence:${item.id}`)) return true;

  if (item.startNodeId && item.nodes?.length) {
    if (!available(item.requires)) return false;

    const reachable = reachableConversationNodes(item);
    if (item.completionEventId &&
        eventAddsNewInfo(item.completionEventId)) return true;

    return item.nodes.some((node) =>
      reachable.has(node.id) &&
      (
        (node.completionEventId &&
          eventAddsNewInfo(node.completionEventId)) ||
        (node.choices || []).some((choice) =>
          available(choice.requires) &&
          choice.completionEventId &&
          eventAddsNewInfo(choice.completionEventId))
      )
    );
  }

  // 旧形式 investigations[] の判定
  return /* 未調査項目から新情報を得られるか */;
}
```

判定条件は、**未閲覧または新規解禁なら表示**。閲覧済みなら、条件を満たす会話・選択肢のEventが未取得の証拠・ログ・判明事項を追加する場合に表示します。Eventによる新情報の判定は、[app.js](/C:/Users/WorkGroup_G4/Documents/cyber-adv-2026/cyber-adv/prototype/app.js:20)にあります。
